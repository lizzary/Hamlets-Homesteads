local prefabs =
{
    "spoiled_food",-- 定义基础预制体依赖列表，所有食物都会包含腐烂食物作为替换
}

local function MakePreparedFood(data)
	local foodassets = -- 创建资源列表，包含基本动画和库存图片
	{
		Asset("ANIM", "anim/cook_pot_food.zip"), -- 基础烹饪锅食物动画
		Asset("INV_IMAGE", data.name),  -- 使用数据中的名称作为库存图片
	}

	if data.overridebuild then -- 如果有自定义构建，添加额外的动画资源
        table.insert(foodassets, Asset("ANIM", "anim/"..data.overridebuild..".zip"))
	end

	local spicename = data.spice ~= nil and string.lower(data.spice) or nil -- 获取香料名称（小写）
    if spicename ~= nil then -- 如果是香料食物，添加香料和盘子相关资源
        table.insert(foodassets, Asset("ANIM", "anim/spices.zip"))
        table.insert(foodassets, Asset("ANIM", "anim/plate_food.zip"))
        table.insert(foodassets, Asset("INV_IMAGE", spicename.."_over"))
    end

    local foodprefabs = prefabs  -- 复制基础预制体列表
    if data.prefabs ~= nil then -- 如果数据有额外预制体依赖，合并到列表中
        foodprefabs = shallowcopy(prefabs)
        for i, v in ipairs(data.prefabs) do
            if not table.contains(foodprefabs, v) then
                table.insert(foodprefabs, v)
            end
        end
    end

    -- 为香料食物创建动态显示名称
    local function DisplayNameFn(inst)
        -- 格式化显示名称（如"香料-食物名"）
        return subfmt(STRINGS.NAMES[data.spice.."_FOOD"], { food = STRINGS.NAMES[string.upper(data.basename)] })
    end

    local function fn() -- 创建预制体
        local inst = CreateEntity()

        inst.entity:AddTransform()
        inst.entity:AddAnimState()
        inst.entity:AddNetwork()

        MakeInventoryPhysics(inst)

		local food_symbol_build = nil  -- 存储食物符号构建名称
        if spicename ~= nil then -- 如果是香料食物
            inst.AnimState:SetBuild("plate_food") -- 设置盘子食物构建
            inst.AnimState:SetBank("plate_food") -- 设置动画银行
            inst.AnimState:OverrideSymbol("swap_garnish", "spices", spicename) -- 覆盖香料符号

            inst:AddTag("spicedfood") -- 添加香料食物标签

            -- 设置物品栏背景
            inst.inv_image_bg = { image = (data.basename or data.name)..".tex" }
            inst.inv_image_bg.atlas = GetInventoryItemAtlas(inst.inv_image_bg.image)

            -- 记录食物符号构建
			food_symbol_build = data.overridebuild or "cook_pot_food"
        else -- 普通食物
			inst.AnimState:SetBuild(data.overridebuild or "cook_pot_food") -- 设置构建
			inst.AnimState:SetBank("cook_pot_food")-- 设置动画
        end

        inst.AnimState:PlayAnimation("idle")
        -- 覆盖食物符号
        inst.AnimState:OverrideSymbol("swap_food", data.overridebuild or "cook_pot_food", data.basename or data.name)
        -- 不知道什么
        inst.scrapbook_overridedata = {"swap_food", data.overridebuild or "cook_pot_food", data.basename or data.name}

        -- 不知道什么
        if data.scrapbook and data.scrapbook.specialinfo then
            inst.scrapbook_specialinfo = data.scrapbook.specialinfo
        end

        inst:AddTag("preparedfood")  -- 添加已准备食物标签

        -- 添加自定义标签
        if data.tags ~= nil then
            for i,v in pairs(data.tags) do
                inst:AddTag(v)
            end
        end

		inst.wet_prefix = data.wet_prefix -- 潮湿前缀（用于潮湿状态显示）

        -- 名称覆盖设置
        if data.basename ~= nil then
            inst:SetPrefabNameOverride(data.basename) -- 设置预制体名称覆盖
            if data.spice ~= nil then
                inst.displaynamefn = DisplayNameFn -- 设置显示名称函数
            end
        end

        if data.floater ~= nil then
            MakeInventoryFloatable(inst, data.floater[1], data.floater[2], data.floater[3])
        else
            MakeInventoryFloatable(inst)
        end

        --不知道是什么
        if data.scrapbook_sanityvalue ~= nil then
            inst.scrapbook_sanityvalue = data.scrapbook_sanityvalue
        end

        if data.scrapbook_healthvalue ~= nil then
            inst.scrapbook_healthvalue = data.scrapbook_healthvalue
        end

        inst.entity:SetPristine()-- 设置实体为原始状态（网络同步用）

        if not TheWorld.ismastersim then
            return inst
        end

		inst.food_symbol_build = food_symbol_build or data.overridebuild-- 存储构建信息
		inst.food_basename = data.basename -- 存储基础名称

        inst:AddComponent("edible") -- 添加可食用组件
        inst.components.edible.healthvalue = data.health
        inst.components.edible.hungervalue = data.hunger
        inst.components.edible.foodtype = data.foodtype or FOODTYPE.GENERIC -- 食物类型
        inst.components.edible.secondaryfoodtype = data.secondaryfoodtype or nil-- 次要食物类型
        inst.components.edible.sanityvalue = data.sanity or 0
        inst.components.edible.temperaturedelta = data.temperature or 0 -- 有的料理吃下去之后会有温度变化
        inst.components.edible.temperatureduration = data.temperatureduration or 0 -- 温度持续时间
        inst.components.edible.nochill = data.nochill or nil -- 是否防止寒冷
        inst.components.edible.spice = data.spice -- 香料类型
        inst.components.edible:SetOnEatenFn(data.oneatenfn) -- 食用后回调函数

        inst:AddComponent("inspectable")

        inst:AddComponent("inventoryitem")
		if data.OnPutInInventory then
			inst:ListenForEvent("onputininventory", data.OnPutInInventory)
		end

        -- 设置物品栏图片
        if spicename ~= nil then
            inst.components.inventoryitem:ChangeImageName(spicename.."_over")
        elseif data.basename ~= nil then
            inst.components.inventoryitem:ChangeImageName(data.basename)
        end

        inst:AddComponent("stackable")
        inst.components.stackable.maxsize = TUNING.STACK_SIZE_SMALLITEM

        if data.perishtime ~= nil and data.perishtime > 0 then
            inst:AddComponent("perishable") -- 添加易腐烂组件
            inst.components.perishable:SetPerishTime(data.perishtime)-- 设置腐烂时间
            inst.components.perishable:StartPerishing()-- 开始腐烂计时
            inst.components.perishable.onperishreplacement = "spoiled_food" -- 腐烂后替换物
        end

        MakeSmallBurnable(inst)
        MakeSmallPropagator(inst)
        MakeHauntableLaunchAndPerish(inst)
        ---------------------

        inst:AddComponent("bait")

        ------------------------------------------------
        inst:AddComponent("tradable")

        ------------------------------------------------

        return inst
    end
    -- NOTES(JBK): Use this to help export the bottom table to make this file findable.
    --print(string.format("%s %s", data.foodtype or FOODTYPE.GENERIC, data.name))
    return Prefab(data.name, fn, foodassets, foodprefabs)
end
